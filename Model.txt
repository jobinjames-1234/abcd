# ============================================================
# 29. EUCLIDEAN DISTANCE
# ============================================================
# WHAT:
#     Measure the straight-line distance between two points.
#
# FORMULA:
#     sqrt(sum((x_i - y_i)^2))
#
# WHY:
#     k-NN uses distance to find nearby observations.
# ============================================================

def euclidean_distance(point1, point2):
    difference = point1 - point2
    squared = difference ** 2
    total = np.sum(squared)

    return np.sqrt(total)


# Small demonstration

)


# ============================================================
# 30. k-NN FROM SCRATCH
# ============================================================
# WHAT:
#     Classify a new point based on its k closest training points.
#
# PROCESS:
#     1. Calculate distances
#     2. Sort distances
#     3. Select k neighbours
#     4. Majority vote
#
# IMPORTANT:
#     This is intentionally implemented manually rather than
#     using KNeighborsClassifier.
# ============================================================

def knn_predict(X_train, y_train, test_point, k=3):

    distances = []

    # Calculate distance to every training point
    for i in range(len(X_train)):

        distance = euclidean_distance(
            X_train[i],
            test_point
        )

        distances.append(
            (distance, y_train.iloc[i])
        )

    # Nearest first
    distances.sort(
        key=lambda x: x[0]
    )

    # Select k nearest
    neighbours = distances[:k]

    # Get their labels
    labels = [
        label
        for distance, label in neighbours
    ]

    # Majority vote
    prediction = max(
        set(labels),
        key=labels.count
    )

    return prediction

# ============================================================
# 31. WEIGHTED k-NN
# ============================================================
# WHAT:
#     Give closer neighbours more influence.
#
# FORMULA:
#     weight = 1 / (distance + epsilon)
#
# WHY:
#     A very close neighbour should matter more than a distant
#     neighbour.
# ============================================================

def weighted_knn_predict(
    X_train,
    y_train,
    test_point,
    k=3,
    epsilon=0.0001
):

    distances = []

    for i in range(len(X_train)):

        distance = euclidean_distance(
            X_train[i],
            test_point
        )

        distances.append(
            (distance, y_train.iloc[i])
        )

    distances.sort(
        key=lambda x: x[0]
    )

    neighbours = distances[:k]

    votes = {}

    for distance, label in neighbours:

        weight = 1 / (distance + epsilon)

        votes[label] = (
            votes.get(label, 0) + weight
        )

    return max(
        votes,
        key=votes.get
    )

        y_train_reset,
        point,
        k=3

# ============================================================
# 34. LAPLACE SMOOTHING
# ============================================================
# WHAT:
#     Prevent zero probabilities.
#
# FORMULA:
#     (count + 1) / (total + number_of_possible_values)
# ============================================================

def laplace_probability(
    count,
    total,
    number_of_categories
):
    return (
        (count + 1)
        /
        (total + number_of_categories)
    )


# Example:
# Suppose a category appears zero times among 9 examples
# and there are 3 possible categories.
zero_probability = 0 / 9

smoothed_probability = laplace_probability(
    count=0,
    total=9,
    number_of_categories=3
)

print("\n" + "=" * 70)
print("34. LAPLACE SMOOTHING")
print("=" * 70)

print("Without smoothing:", zero_probability)
print("With smoothing:", smoothed_probability)

# ============================================================
# 37. ENTROPY FROM SCRATCH
# ============================================================
# WHAT:
#     Measure impurity/uncertainty in a set of class labels.
#
# FORMULA:
#     Entropy = -sum(p * log2(p))
#
# Pure data -> entropy 0
# Mixed data -> higher entropy
# ============================================================

def entropy(labels):

    # Count occurrences of each class
    counts = pd.Series(labels).value_counts()

    total = len(labels)

    result = 0.0

    for count in counts:

        probability = count / total

        # Ignore zero because log2(0) is undefined.
        if probability > 0:

            result -= (
                probability
                * math.log2(probability)
            )

    return result


example_labels = [
    "Yes",
    "Yes",
    "No",
    "No"
]

print("\n" + "=" * 70)
print("37. ENTROPY")
print("=" * 70)

print(
    "Entropy of [Yes, Yes, No, No]:",
    entropy(example_labels)
)


# ============================================================
# 38. INFORMATION GAIN FROM SCRATCH
# ============================================================
# WHAT:
#     Measure how much a split reduces entropy.
#
# FORMULA:
#
#     Information Gain
#       = Parent Entropy
#         - Weighted Child Entropy
#
# Higher gain = more useful split.
# ============================================================

def information_gain(parent_labels, child_groups):

    parent_entropy = entropy(parent_labels)

    total = len(parent_labels)

    weighted_child_entropy = 0.0

    for child in child_groups:

        if len(child) == 0:
            continue

        weight = len(child) / total

        weighted_child_entropy += (
            weight * entropy(child)
        )

    gain = (
        parent_entropy
        - weighted_child_entropy
    )

    return gain


parent = [
    "Yes", "Yes", "Yes",
    "No", "No", "No"
]

    ["No", "No", "No"]

print("38. INFORMATION GAIN")

)

)
